DECLARE
  v_count NUMBER;

  PROCEDURE add_column(
    p_table VARCHAR2,
    p_column VARCHAR2,
    p_definition VARCHAR2
  ) IS
    v_column_count NUMBER;
  BEGIN
    SELECT COUNT(*)
    INTO v_column_count
    FROM user_tab_columns
    WHERE table_name = UPPER(p_table)
      AND column_name = UPPER(p_column);

    IF v_column_count = 0 THEN
      EXECUTE IMMEDIATE
        'ALTER TABLE ' || p_table ||
        ' ADD (' || p_column || ' ' || p_definition || ')';

      DBMS_OUTPUT.PUT_LINE('Added ' || p_table || '.' || p_column);
    ELSE
      DBMS_OUTPUT.PUT_LINE(
        'Already exists: ' || p_table || '.' || p_column
      );
    END IF;
  END;
BEGIN
  add_column(
    'APPLICATIONS',
    'APPLICATION_NUMBER',
    'VARCHAR2(30)'
  );

  add_column(
    'APPLICATIONS',
    'SUBMITTED_AT',
    'TIMESTAMP WITH TIME ZONE'
  );

  add_column(
    'APPLICATIONS',
    'REVIEWED_AT',
    'TIMESTAMP WITH TIME ZONE'
  );

  add_column(
    'APPLICATIONS',
    'REVIEWED_BY',
    'VARCHAR2(100)'
  );

  add_column(
    'APPLICATIONS',
    'REVIEW_NOTE',
    'VARCHAR2(2000)'
  );

  add_column(
    'DOCUMENTS',
    'SIZE_BYTES',
    'NUMBER'
  );
END;
/

BEGIN
  SELECT COUNT(*)
  INTO v_exists
  FROM user_constraints
  WHERE table_name = 'APPLICATIONS'

  IF v_exists = 0 THEN
    EXECUTE IMMEDIATE
